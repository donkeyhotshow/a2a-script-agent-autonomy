Проверьте следующий JSON на валидность и соответствие схеме:

```json
{
    "name": "test",
    "config": {
        "enabled": true,
        "settings": {
            "timeout": 30,
            "retries": 3
        }
    },
    "dependencies": [
        {
            "id": "1",
            "version": "1.0"
        }
    ]
}
```

Схема:
```json
{
    "type": "object",
    "required": ["name", "config", "dependencies"],
    "properties": {
        "name": {
            "type": "string"
        },
        "config": {
            "type": "object",
            "required": ["enabled", "settings"],
            "properties": {
                "enabled": {
                    "type": "boolean"
                },
                "settings": {
                    "type": "object",
                    "required": ["timeout", "retries"],
                    "properties": {
                        "timeout": {
                            "type": "integer",
                            "minimum": 0
                        },
                        "retries": {
                            "type": "integer",
                            "minimum": 0
                        }
                    }
                }
            }
        },
        "dependencies": {
            "type": "array",
            "items": {
                "type": "object",
                "required": ["id", "version"],
                "properties": {
                    "id": {
                        "type": "string"
                    },
                    "version": {
                        "type": "string"
                    }
                }
            }
        }
    }
}
```
