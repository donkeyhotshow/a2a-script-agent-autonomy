```powershell
function Resolve-SimpleTemplate {
    [CmdletBinding()]
    param(
        [string]$TemplateString,
        [hashtable]$Context
    )
    $resolvedString = $TemplateString

    if ($null -ne $Context) {
        # Step 1 & 2: Handle conditional blocks first
        if ($resolvedString -match '\{\{#if context\.taskFinalized\}\}') {
            if ($Context.taskFinalized -eq $true) {
                # Keep the 'then' part, discard 'else' part and the #if/#else/#endif tags
                $resolvedString = $resolvedString -replace '\{\{#if context\.taskFinalized\}\}(.*?)\{\{else\}\}.*?\{\{/if\}\}', '$1'
            } else {
                # Keep the 'else' part, discard 'then' part and the #if/#else/#endif tags
                $resolvedString = $resolvedString -replace '\{\{#if context\.taskFinalized\}\}.*?\{\{else\}\}(.*?)\{\{/if\}\}', '$1'
            }
        }
        
        if ($resolvedString -match '\{\{#if context\.finalMessage\}\}') {
            if ($Context.finalMessage -and $Context.finalMessage -ne '') {
                # Keep the 'then' part, discard the #if/#endif tags

            } else {
                # Discard the whole #if block
                $resolvedString = $resolvedString -replace '\{\{#if context\.finalMessage\}\}.*?\{\{/if\}\}', ''
            }
        }

        # Step 3: Resolve all remaining {{context.key}} and {{key}} placeholders
        foreach ($key in $Context.Keys) {
            $placeholder = "{{context.$key}}"
            if ($resolvedString -match [regex]::Escape($placeholder)) {
                $resolvedString = $resolvedString -replace [regex]::Escape($placeholder), $Context[$key]
            }
            $placeholderSimple = "{{$key}}"
            if ($resolvedString -match [regex]::Escape($placeholderSimple)) {
                $resolvedString = $resolvedString -replace [regex]::Escape($placeholderSimple), $Context[$key]
            }
        }
    }
    return $resolvedString
}
```
