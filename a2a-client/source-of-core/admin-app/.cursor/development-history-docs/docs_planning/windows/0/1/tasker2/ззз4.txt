```powershell
[CmdletBinding()]
param (
    [Parameter(Mandatory=$true)]
    [hashtable]$ScenarioContext, # Контекст текущего сценария

    [Parameter(Mandatory=$true)]
    [string]$QuestionText,      # Текст вопроса

    [Parameter(Mandatory=$true)]
    [string]$AnswerKey,         # Ключ для сохранения ответа в контексте

    [string]$QuestionCategory = "ScenarioInput",
    [string]$QuestionType = "text", # Тип ожидаемого ответа (text, choice, etc.)
    [hashtable]$QuestionContextSchema = @{ type = "string" }, # Схема для валидации/отображения
    
    [string]$QuestionsFilePath = "./script/engine/QandA/questions.json"
)

try {
    $questionId = "QID-$(Get-Date -Format 'yyyyMMddHHmmssffff')-$([System.Guid]::NewGuid().ToString().Substring(0,4))"

    $newQuestion = @{
        id = $questionId
        category = $QuestionCategory
        type = $QuestionType
        text_ru = $QuestionText
        sourceTaskId = $ScenarioContext.sourceTaskId # Предполагается, что это может быть в контексте
        sourceScenarioId = $ScenarioContext.scenarioId # ID текущего сценария
        sourceStepId = $ScenarioContext.currentStepId # ID текущего шага сценария, который вызывает этот скрипт
        status = "pending_answer"
        contextSchema = $QuestionContextSchema
        answerKey = $AnswerKey
        # timestamp_generated - убрали по просьбе пользователя
    }

    $questions = @()
    if (Test-Path $QuestionsFilePath) {
        try {
            $questions = Get-Content $QuestionsFilePath -Raw -ErrorAction Stop | ConvertFrom-Json -ErrorAction Stop
            if ($null -eq $questions -or $questions -isnot [System.Array]) { # если файл пуст или не массив
                 $questions = @()
            }
        } catch {
            Write-Warning "Error reading or parsing '$QuestionsFilePath': $($_.Exception.Message). Initializing as empty array."
            $questions = @()
        }
    }

    $questions += $newQuestion
    $questions | ConvertTo-Json -Depth 5 | Set-Content -Path $QuestionsFilePath -Force

    # Возвращаем ID созданного вопроса для помещения в контекст сценария
    Write-Output @{ pendingQuestionId = $questionId }

} catch {
    Write-Error "Error in GenerateQuestion.ps1: $($_.Exception.Message)"
    throw # Перебрасываем исключение, чтобы сценарий мог его обработать
}
```

**Пояснения к скрипту:**
*   Принимает параметры: контекст сценария, текст вопроса, ключ для ответа и опциональные детали.
*   Генерирует `questionId` с использованием текущей даты/времени и части GUID для уникальности.
*   Формирует объект вопроса, включая поля `sourceScenarioId` и `sourceStepId` из `$ScenarioContext` (предполагается, что `Invoke-ScenarioEngine.ps1` передает эти данные в `$ScenarioContext` при вызове скрипта действия).
*   Безопасно читает и обновляет `questions.json`. Если файл не существует, пуст или поврежден, он инициализируется как пустой массив.
*   Возвращает `@{ pendingQuestionId = $questionId }`, чтобы движок сценариев мог обновить свой контекст.

```powershell
[CmdletBinding()]
param (
    [Parameter(Mandatory=$true)]
    [hashtable]$ScenarioContext,      # Контекст текущего сценария

    [string]$QuestionsFilePath = "./script/engine/QandA/questions.json",
    [string]$AnswersFilePath = "./script/engine/QandA/answers.json"
)

try {
    if (-not $ScenarioContext.ContainsKey("pendingQuestionId") -or [string]::IsNullOrEmpty($ScenarioContext.pendingQuestionId)) {
        Write-Warning "GetAnswerAndUpdateContext.ps1: 'pendingQuestionId' not found or empty in ScenarioContext. Cannot proceed."
        # Возвращаем результат, который не изменит контекст и покажет, что нужно действие
        Write-Output @{ needsUserInput = $true; statusMessage = "Waiting for pendingQuestionId to be set." }
        return
    }

    $pendingQuestionId = $ScenarioContext.pendingQuestionId
    $answerFound = $null
    $answerValue = $null
    $returnContextUpdate = @{ pendingQuestionId = $pendingQuestionId } # По умолчанию сохраняем pendingQuestionId

    if (Test-Path $AnswersFilePath) {
        $answers = Get-Content $AnswersFilePath -Raw -ErrorAction SilentlyContinue | ConvertFrom-Json -ErrorAction SilentlyContinue
        if ($null -ne $answers -and $answers -is [System.Array]) {
            $answerFound = $answers | Where-Object { $_.questionId -eq $pendingQuestionId } | Select-Object -First 1
        }
    }

    if ($null -ne $answerFound) {
        $answerValue = $answerFound.answerValue
        $answeredBy = $answerFound.answeredBy

        # Найти answerKey в questions.json
        $questions = Get-Content $QuestionsFilePath -Raw | ConvertFrom-Json
        $questionOrigin = $questions | Where-Object { $_.id -eq $pendingQuestionId } | Select-Object -First 1

        if ($null -eq $questionOrigin) {
            Write-Warning "Original question with ID '$pendingQuestionId' not found in '$QuestionsFilePath'."
            # Продолжаем без answerKey, но это может быть проблемой для сценария
            $returnContextUpdate[$pendingQuestionId] = $answerValue # Используем сам ID как ключ, если answerKey не найден
        } else {
            $answerKey = $questionOrigin.answerKey
            if ([string]::IsNullOrEmpty($answerKey)) {
                Write-Warning "AnswerKey is empty for question ID '$pendingQuestionId'. Using question ID as context key."
                $returnContextUpdate[$pendingQuestionId] = $answerValue
            } else {
                $returnContextUpdate[$answerKey] = $answerValue
            }
            
            # Обновить статус вопроса в questions.json
            $questionOrigin.status = "answered_by_user" # Или можно использовать $answeredBy
            # $questionOrigin.timestamp_answered - убрали
            $questions | ConvertTo-Json -Depth 5 | Set-Content -Path $QuestionsFilePath -Force
        }
        
        $returnContextUpdate.Remove("pendingQuestionId") # Удаляем, так как вопрос отвечен
        $returnContextUpdate["needsUserInput"] = $false
        $returnContextUpdate["statusMessage"] = "Answer processed for $pendingQuestionId."
        
        Write-Output $returnContextUpdate

    } else {
        # Ответ еще не предоставлен
        # Сигнализируем, что сценарий должен ждать
        Write-Output @{ needsUserInput = $true; statusMessage = "Waiting for answer to question ID: $pendingQuestionId" }
    }

} catch {
    Write-Error "Error in GetAnswerAndUpdateContext.ps1: $($_.Exception.Message)"
    # В случае ошибки, также сигнализируем о необходимости вмешательства или проверки
    Write-Output @{ needsUserInput = $true; statusMessage = "Error processing answer: $($_.Exception.Message)"; errorOccurred = $true }
    throw
}
```

**Пояснения к скрипту:**
*   Проверяет наличие `pendingQuestionId` в контексте.
*   Ищет ответ в `answers.json`.
*   Если ответ найден:
    *   Извлекает `answerKey` из оригинального вопроса в `questions.json`.
    *   Готовит хеш-таблицу для обновления контекста сценария (`@{ <answerKey> = <answerValue> }`).
    *   Обновляет статус вопроса в `questions.json`.
    *   Удаляет `pendingQuestionId` из возвращаемого обновления (сигнал, что этот вопрос закрыт).
    *   Устанавливает `needsUserInput = $false`.
*   Если ответ не найден:
    *   Возвращает `@ needsUserInput = $true; statusMessage = "Waiting..." }`. Это позволит `Invoke-ScenarioEngine.ps1` понять, что нужно прервать выполнение текущего сценария и ждать внешнего события (пользователь добавит ответ). Движок сценариев не должен переходить к следующему шагу, если `needsUserInput` вернулся как `$true`.

**Важно**: Логика `Invoke-ScenarioEngine.ps1` должна быть готова обрабатывать такой вывод от скриптов действий. Если скрипт возвращает объект с `needsUserInput = $true`, движок должен приостановить выполнение текущего сценария, сохранить его состояние (включая `pendingQuestionId` в контексте) и не переходить к `nextStepId`. `main-work.ps1` затем сможет информировать AI о том, что система ожидает ответа на конкретный вопрос.
